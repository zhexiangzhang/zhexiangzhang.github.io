<script>
document.addEventListener("DOMContentLoaded", function () {
  const timeline = document.querySelector(".timeline-v2");
  if (!timeline || timeline.dataset.enhanced === "true") return;

  timeline.dataset.enhanced = "true";
  const monthNames = {
    January: "Jan", February: "Feb", March: "Mar", April: "Apr",
    May: "May", June: "Jun", July: "Jul", August: "Aug",
    September: "Sep", October: "Oct", November: "Nov", December: "Dec"
  };
  let currentYear = "";

  timeline.querySelectorAll(":scope > .timeline-item").forEach(function (item) {
    const header = item.querySelector(".timeline-header");
    const date = item.querySelector(".timeline-date");
    const place = item.querySelector(".timeline-place");
    if (!header || !date || !place) return;

    const dateParts = date.textContent.trim().match(/^(\w+)\s+(\d{4})$/);
    if (dateParts) {
      const month = dateParts[1];
      const year = dateParts[2];
      if (year !== currentYear) {
        const yearHeading = document.createElement("div");
        yearHeading.className = "timeline-year";
        yearHeading.textContent = year;
        timeline.insertBefore(yearHeading, item);
        currentYear = year;
      }
      date.textContent = monthNames[month] || month.slice(0, 3);
    }

    const placeParts = place.textContent.trim().split(", ");
    const cityName = placeParts.shift() || "";
    const countryName = placeParts.join(", ");
    const city = document.createElement("span");
    city.className = "timeline-city";
    city.textContent = cityName;
    const country = document.createElement("span");
    country.className = "timeline-country";
    country.textContent = countryName || cityName;
    if (countryName) {
      place.replaceChildren(country, city);
    } else {
      place.replaceChildren(country);
    }

    const chevron = document.createElement("span");
    chevron.className = "timeline-chevron";
    chevron.setAttribute("aria-hidden", "true");
    chevron.textContent = "⌄";
    header.appendChild(chevron);
    header.setAttribute("role", "button");
    header.setAttribute("tabindex", "0");
    header.setAttribute("aria-expanded", "false");

    function toggleItem() {
      const isOpen = item.classList.toggle("open");
      header.setAttribute("aria-expanded", String(isOpen));
    }

    header.addEventListener("click", toggleItem);
    header.addEventListener("keydown", function (event) {
      if (event.key === "Enter" || event.key === " ") {
        event.preventDefault();
        toggleItem();
      }
    });
  });
});
</script>
